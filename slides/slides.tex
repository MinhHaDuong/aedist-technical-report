\section{Problem}

% -------------------------------------------------------------------
\begin{frame}{Energy statistics where offices are weak}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{Challenge:} Track energy transition in countries lacking strong statistical offices

\medskip
\textbf{Case study:} Vietnam thermal power plants
\begin{itemize}
  \item 164 plants in reference inventory
  \item Coal, gas, LNG -- operational to proposed
  \item Expert-maintained dataset
\end{itemize}

\medskip
\textbf{Question:} Can AI automate this?

\column{0.42\textwidth}
\textbf{Why it's hard:}
\begin{itemize}
  \item \textbf{Multilingual} -- Vietnamese diacritics, transliterations
  \item \textbf{Temporal} -- plans change, plants retire
  \item \textbf{Technical} -- kW vs kWh, MWth vs MWe
  \item \textbf{Granularity} -- complex vs plant vs unit vs turbine
\end{itemize}
\end{columns}
\end{frame}

% ===================================================================
\section{Benchmark design}

% -------------------------------------------------------------------
\begin{frame}[fragile]{The Vietnam thermal plant benchmark}
\begin{columns}[T]
\column{0.52\textwidth}
\textbf{Prompt} (verbatim):
\begin{quote}\small\itshape
Provide a table listing all thermal power plants in Vietnam, including their fuel type (coal / local natural gas / imported LNG), construction stage, connection date (observed or planned COD), province, and generation capacity (in MWe). Format the response in CSV.
\end{quote}

\medskip
\textbf{Reference:} 163 thermal power plants\\
6 columns: name, fuel, status, COD, province, capacity\\
Expert-validated from PDP7, PDP8, EVN reports

\column{0.45\textwidth}
\textbf{Experimental design:}
\begin{description}
  \item[Purpose] Measure LLM recall of structured energy data
  \item[Apparatus] OpenRouter API + local Ollama
  \item[N] 28 cloud + 8 local models $\times$ 3 runs = 108
  \item[Cost] \$0.73 total
  \item[Evaluation] Exact + fuzzy name matching, then per-field accuracy
\end{description}

\medskip
\textbf{Metric:} plants matched out of 163\\
{\small (precision $>$99\% across all models --- the main\\problem is \textbf{coverage}, not hallucination)}
\end{columns}
\end{frame}

\section{Verification}

% -------------------------------------------------------------------
\begin{frame}{Can we trust AI statistics?}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Error taxonomy:}
\begin{description}
  \item[Hallucinated] plants that don't exist
  \item[Missed] real plants not returned
  \item[Wrong fuel] coal vs gas confusion
  \item[Wrong status] operational vs planned
  \item[Wrong province] location errors
  \item[Capacity mismatch] wrong MW value
\end{description}

\column{0.48\textwidth}
\textbf{Reconciliation pipeline:}
\begin{enumerate}
  \item Exact name match
  \item Fuzzy match (diacritics, transliterations)
  \item Per-field accuracy on matched plants
\end{enumerate}

\medskip
\textbf{Key finding:}\\
Attribute accuracy on matched plants is high ($>$90\% for fuel, province, capacity).\\
The main problem is \textbf{coverage}, not accuracy.
\end{columns}
\end{frame}

\section{Architecture}

% -------------------------------------------------------------------
\begin{frame}{AEDIST: proposed architecture}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Four information layers:}
\begin{enumerate}
  \item \textbf{Document store} -- cached web files
  \item \textbf{Validated base} -- Zotero-indexed, expert-reviewed
  \item \textbf{Vector layer} -- semantic search for RAG
  \item \textbf{Knowledge graph} -- Open Energy Ontology, PROV traceability
\end{enumerate}

\column{0.48\textwidth}
\textbf{Specialized agents:}
\begin{description}
  \item[Watcher] source monitoring, classification
  \item[Curator] indexing, metadata normalization
  \item[Analyst] entity extraction, graph construction
  \item[Writer] structured reports, data export
\end{description}

\medskip
\textbf{Key principle:}\\
Accumulation over time, not instant search.\\
Human validates before ingestion.
\end{columns}
\end{frame}

\section{Takeaway}

% -------------------------------------------------------------------
\begin{frame}{Summary of findings}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{What works:}
\begin{itemize}
  \item RAG + conversational follow-ups
  \item PDF $\to$ Markdown conversion
  \item OpenStreetMap as complementary source
  \item Automated evaluation pipeline
\end{itemize}

\medskip
\textbf{What's needed:}
\begin{itemize}
  \item Local knowledge graph (OEO-aligned)
  \item Specialized agents (curator, analyst)
  \item Four-layer architecture
\end{itemize}

\column{0.48\textwidth}
\textbf{Key numbers:}
\begin{itemize}
  \item 28 cloud + 8 local models, \$0.73 census
  \item Direct: best 107/163 plants (GPT-5.4)
  \item RAG: best 141/163 plants (Mistral Small 4)
  \item Attribute accuracy $>$90\% on matches
\end{itemize}

\medskip
\textbf{Bottom line:}\\
Hybrid approach -- RAG + knowledge graphs + specialized agents -- is \textbf{necessary and feasible} for traceable energy statistics from heterogeneous, multilingual sources.
\end{columns}
\end{frame}
